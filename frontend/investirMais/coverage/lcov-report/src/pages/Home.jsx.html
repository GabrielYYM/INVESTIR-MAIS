function formatDateBr(isoStr) {
  if (!isoStr) return "";
  const parts = String(isoStr).split("-");
  if (parts.length !== 3) return isoStr;
  return `${parts[2]}/${parts[1]}/${parts[0]}`;
}
 
function toIsoDate(d) {
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}
 
function validatePeriod(startStr, endStr) {
  if (!startStr || !endStr) return { valid: false, error: "Informe as datas de início e fim." };
  const start = new Date(startStr + "T00:00:00");
  const end = new Date(endStr + "T00:00:00");
  if (isNaN(start.getTime()) || isNaN(end.getTime())) {
    return { valid: false, error: "Data em formato inválido." };
  }
  if (start > end) {
    return { valid: false, error: "A data inicial não pode ser posterior à data final." };
  }
  const maxEnd = new Date(start);
  maxEnd.setMonth(maxEnd.getMonth() + 6);
  if (end > maxEnd) {
    return {
      valid: false,
      error: "O período de consulta não pode exceder 6 meses para proteger a performance da plataforma.",
    };
  }
  return { valid: true, error: null };
}

export async function quoteLoader({ request }) {
  const ticker = new URL(request.url).searchParams.get("ticker")?.trim().toUpperCase();
  if (!ticker || !/^[A-Z0-9.-]{1,16}$/.test(ticker)) return { error: "Informe um ticker válido." };
 
  try {
    const data = await buscarCotacao(ticker, request.signal);
    const quote = data.results?.[0];
    if (!quote || !Number.isFinite(quote.regularMarketPrice)) return { error: "Nenhuma cotação disponível para " + ticker + "." };
    return { quote };
  } catch (error) {
    if (error.response?.status === 401) throw redirect("/entrar");
    return {
      error: error.response?.status === 404
        ? "Ativo não encontrado. Confira o ticker informado."
        : "A cotação está indisponível no momento. Tente novamente.",
    };
  }
}
